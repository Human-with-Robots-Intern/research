# Python ai2thor control script
# A list of available functions. The LLM must use the function names and formats from this list.
# Available functions: move_to, pickup, slice, say, put, drop, toggle_on, toggle_off, open, close, is_obj_visible, get_obj_names, get_obj_id, parse_question, wait

# [RULES]
# - Making tomato sauce: ALWAYS place tomato inside blue_pot first, then place blue_pot on stove. Never use pan for tomato sauce.
# - Cooking sausage: use pan on stove.
# - Making tea: place tea_cup inside tea_pot, then toggle_on tea_pot.

objects = ['banana', 'plate', 'pan', 'sausage', 'stove', 'orange_bowl', 'purple_bowl', 'sink|sinkbasin', 'tea_pot', 'tea_cup', 'green_cup', 'tomato', 'table', 'carrot', 'blue_pot',]
# cook sausage in a pan on the stove, make tomato sauce using blue_pot, and make tea using the tea_pot.
say('OK - cook sausage, make tomato sauce, and make tea')
# Task 1: Cook Sausage — sausage goes into the pan
sausage_id = get_obj_id('sausage')
pan_id = get_obj_id('pan')
if sausage_id and pan_id:
    move_to(sausage_id)
    if parse_question('is anything in hand?'):
        drop()
    pickup(sausage_id)
    move_to(pan_id)
    put(pan_id)
else:
    say("I can't find sausage or pan.")
# put pan on the stove
pan_id = get_obj_id('pan')
stove_id = get_obj_id('stove')
if pan_id and stove_id:
    move_to(pan_id)
    if parse_question('is anything in hand?'):
        drop()
    pickup(pan_id)
    move_to(stove_id)
    put(stove_id)
else:
    say("I can't find pan or stove.")
# turn on stove and wait for sausage to cook
stove_id = get_obj_id('stove')
if stove_id:
    move_to(stove_id)
    toggle_on(stove_id)
    wait(10)
    toggle_off(stove_id)
else:
    say("I can't find the stove.")
# Task 2: Make Tomato Sauce — tomato sauce requires blue_pot, NOT pan
tomato_id = get_obj_id('tomato')
blue_pot_id = get_obj_id('blue_pot')
if tomato_id and blue_pot_id:
    move_to(tomato_id)
    if parse_question('is anything in hand?'):
        drop()
    pickup(tomato_id)
    move_to(blue_pot_id)
    put(blue_pot_id)
else:
    say("I can't find tomato or blue_pot.")
# put blue_pot on the stove
blue_pot_id = get_obj_id('blue_pot')
stove_id = get_obj_id('stove')
if blue_pot_id and stove_id:
    move_to(blue_pot_id)
    if parse_question('is anything in hand?'):
        drop()
    pickup(blue_pot_id)
    move_to(stove_id)
    put(stove_id)
else:
    say("I can't find blue_pot or stove.")
# turn on stove and wait for tomato sauce
stove_id = get_obj_id('stove')
if stove_id:
    move_to(stove_id)
    toggle_on(stove_id)
    wait(10)
    toggle_off(stove_id)
else:
    say("I can't find the stove.")
# Task 3: Make Tea
# put tea_cup into tea_pot
tea_cup_id = get_obj_id('tea_cup')
tea_pot_id = get_obj_id('tea_pot')
if tea_cup_id and tea_pot_id:
    move_to(tea_cup_id)
    if parse_question('is anything in hand?'):
        drop()
    pickup(tea_cup_id)
    move_to(tea_pot_id)
    put(tea_pot_id)
else:
    say("I can't find tea_cup or tea_pot.")
# turn on tea_pot and wait for tea to be ready
tea_pot_id = get_obj_id('tea_pot')
if tea_pot_id:
    move_to(tea_pot_id)
    toggle_on(tea_pot_id)
    wait(10)
    toggle_off(tea_pot_id)
else:
    say("I can't find the tea_pot.")
