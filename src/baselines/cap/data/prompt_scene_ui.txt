import numpy as np
import time
# A list of available functions. The LLM must use the function names and formats from this list.
# Available functions: move_to, pickup, slice, say, put,  drop, toggle_on, toggle_off, open, close, is_obj_visible, get_obj_names, get_obj_id
from utils import move_to, pickup, slice, say, put,  drop, toggle_on, toggle_off, open, close, is_obj_visible, get_obj_names, get_obj_id
from plan_utils import parse_question


objects = ['banana', 'plate', 'pan', 'sausage', 'stove', 'orange_bowl', 'purple_bowl', 'sink|sinkbasin', 'tea_pot', 'tea_cup', 'green_cup', 'tomato', 'table', 'carrot', 'blue_pot',]
# cook sausage in a pan on the stove.
say('OK - cook sausage')
# first put sausage into the pan
sausage_id = get_obj_id('sausage')
pan_id = get_obj_id('pan')
if sausage_id and pan_id:
    move_to(sausage_id)
    if parse_question('is anything in hand?'):
        drop()
    pickup(sausage_id)
    move_to(pan_id)
    put(pan_id)
else:
    say("I can't find sausage or pan.")
# put pan on the stove
pan_id = get_obj_id('pan')
stove_id = get_obj_id('stove')
if pan_id and stove_id:
    move_to(pan_id)
    if parse_question('is anything in hand?'):
        drop()
    pickup(pan_id)
    move_to(stove_id)
    put(stove_id)
else:
    say("I can't find pan or stove.")
# turn on stove and wait for cooking
stove_id = get_obj_id('stove')
if stove_id:
    move_to(stove_id)
    toggle_on(stove_id)
    wait(10)
    toggle_off(stove_id)
else:
    say("I can't find the stove.")
