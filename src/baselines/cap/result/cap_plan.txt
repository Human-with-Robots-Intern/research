objects = ['Apple', 'DishSponge', 'Plate', 'Bottle', 'Potato', 'Egg', 'Stool', 'Fridge', 'CreditCard', 'Pot', 'Tomato', 'PaperTowelRoll', 'Bread', 'Kettle', 'Toaster', 'Book', 'Pan', 'Spoon', 'Drawer', 'Floor', 'Knife', 'Faucet', 'PepperShaker', 'CoffeeMachine', 'SinkBasin', 'HousePlant', 'SaltShaker', 'GarbageCan', 'ButterKnife', 'ShelvingUnit', 'Microwave', 'CounterTop', 'Lettuce', 'Cup', 'Cabinet', 'Sink', 'StoveBurner', 'Mug', 'Vase', 'WineBottle', 'Statue', 'Bowl', 'Window', 'Fork', 'Shelf', 'StoveKnob', 'LightSwitch', 'Spatula', 'SoapBottle']

# Boil Potato
potato_id = get_obj_id('Potato')
pot_id = get_obj_id('Pot')
stoveburner_id = get_obj_id('StoveBurner')
if potato_id and pot_id:
    move_to(potato_id)
    pickup(potato_id)
    move_to(pot_id)
    put(pot_id)
    move_to(stoveburner_id)
    pickup(pot_id)
    put(stoveburner_id)
    stoveknob_id = get_obj_id('StoveKnob')
    toggle_on(stoveknob_id)
    wait(100)
    toggle_off(stoveknob_id)
else:
    say("I can't find the potato or pot.")

# Cook Egg
egg_id = get_obj_id('Egg')
pan_id = get_obj_id('Pan')
if egg_id and pan_id:
    move_to(egg_id)
    pickup(egg_id)
    move_to(pan_id)
    put(pan_id)
    move_to(stoveburner_id)
    pickup(pan_id)
    put(stoveburner_id)
    toggle_on(stoveknob_id)
    wait(100)
    toggle_off(stoveknob_id)
else:
    say("I can't find the egg or pan.")

# Wash all Fork and Spoon
fork_id = get_obj_id('Fork')
spoon_id = get_obj_id('Spoon')
sinkbasin_id = get_obj_id('SinkBasin')
faucet_id = get_obj_id('Faucet')

if fork_id:
    move_to(fork_id)
    pickup(fork_id)
    move_to(sinkbasin_id)
    toggle_on(faucet_id)
    put(sinkbasin_id)
    toggle_off(faucet_id)
    pickup(fork_id)
    move_to(get_obj_id('CounterTop'))
    put(get_obj_id('CounterTop'))
else:
    say("I can't find the fork.")

if spoon_id:

    pickup(spoon_id)
    move_to(sinkbasin_id)
    toggle_on(faucet_id)
    put(sinkbasin_id)
    toggle_off(faucet_id)
    pickup(spoon_id)
    move_to(get_obj_id('CounterTop'))
    put(get_obj_id('CounterTop'))
else:
    say("I can't find the spoon.")
