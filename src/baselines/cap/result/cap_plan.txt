# make a coffee
mug_id = get_obj_id('mug')
if mug_id:
    move_to(mug_id)
    pickup(mug_id)
    coffee_machine_id = get_obj_id('coffeemachine')
    move_to(coffee_machine_id)
    put(coffee_machine_id)
    toggle_on(coffee_machine_id)
    if not parse_question('is mug filled with coffee?'):
        toggle_on(coffee_machine_id)
    pickup(mug_id)
    countertop_id = get_obj_id('countertop')
    move_to(countertop_id)
    put(countertop_id)
else:
    say("I can't find the mug.")
