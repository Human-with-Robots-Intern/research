# 07_make_tomato_sauce_and_make_tea_in_teapot_and_put_green_cup_in_sink.json.
# say that you will make tomato sauce and tea, and put green cup in sink
say('OK - make tomato sauce, make tea, and put green cup in sink')

# Task 1: Make Tomato Sauce — tomato sauce requires blue_pot, NOT pan
tomato_id = get_obj_id('tomato')
blue_pot_id = get_obj_id('blue_pot')
if tomato_id and blue_pot_id:
    move_to(tomato_id)
    if parse_question('is anything in hand?'):
        drop()
    pickup(tomato_id)
    move_to(blue_pot_id)
    put(blue_pot_id)
else:
    say("I can't find tomato or blue_pot.")

# put blue_pot on the stove
blue_pot_id = get_obj_id('blue_pot')
stove_id = get_obj_id('stove')
if blue_pot_id and stove_id:
    move_to(blue_pot_id)
    if parse_question('is anything in hand?'):
        drop()
    pickup(blue_pot_id)
    move_to(stove_id)
    put(stove_id)
else:
    say("I can't find blue_pot or stove.")

# turn on stove and wait for tomato sauce
stove_id = get_obj_id('stove')
if stove_id:
    move_to(stove_id)
    toggle_on(stove_id)
    wait(10)
    toggle_off(stove_id)
else:
    say("I can't find the stove.")

# Task 2: Make Tea
# put tea_cup into tea_pot
tea_cup_id = get_obj_id('tea_cup')
tea_pot_id = get_obj_id('tea_pot')
if tea_cup_id and tea_pot_id:
    move_to(tea_cup_id)
    if parse_question('is anything in hand?'):
        drop()
    pickup(tea_cup_id)
    move_to(tea_pot_id)
    put(tea_pot_id)
else:
    say("I can't find tea_cup or tea_pot.")

# turn on tea_pot and wait for tea to be ready
tea_pot_id = get_obj_id('tea_pot')
if tea_pot_id:
    move_to(tea_pot_id)
    toggle_on(tea_pot_id)
    wait(10)
    toggle_off(tea_pot_id)
else:
    say("I can't find the tea_pot.")

# Task 3: Put Green Cup in Sink
green_cup_id = get_obj_id('green_cup')
sink_id = get_obj_id('sink|sinkbasin')
if green_cup_id and sink_id:
    move_to(green_cup_id)
    if parse_question('is anything in hand?'):
        drop()
    pickup(green_cup_id)
    move_to(sink_id)
    put(sink_id)
else:
    say("I can't find green_cup or sink.")
