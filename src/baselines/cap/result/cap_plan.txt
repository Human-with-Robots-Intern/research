# make a coffee
mug_id = get_obj_id('mug')
if mug_id:
    move_to(mug_id)
    pickup(mug_id)
    coffee_machine_id = get_obj_id('coffeemachine')
    move_to(coffee_machine_id)
    put(coffee_machine_id)
    toggle_on(coffee_machine_id)
    if not parse_question('is mug filled with coffee?'):
        toggle_on(coffee_machine_id)
    pickup(mug_id)
    countertop_id = get_obj_id('countertop')
    move_to(countertop_id)
    put(countertop_id)
else:
    say("I can't find the mug.")

# wash apple and lettuce
for item in ['apple', 'lettuce']:
    item_id = get_obj_id(item)
    if item_id:
        move_to(item_id)
        pickup(item_id)
        sinkbasin_id = get_obj_id('sinkbasin')
        move_to(sinkbasin_id)
        faucet_id = get_obj_id('faucet')
        toggle_on(faucet_id)
        if parse_question(f'isn\'t {item} in hand?'):
            move_to(item_id)
            pickup(item_id)
            move_to(sinkbasin_id)
        put(sinkbasin_id)
        toggle_off(faucet_id)
        pickup(item_id)
        move_to(countertop_id)
        if parse_question(f'isn\'t {item} in hand?'):
            move_to(item_id)
            pickup(item_id)
            move_to(countertop_id)
        put(countertop_id)
    else:
        say(f"I can't find the {item}.")
